rule no-module-cycles {
  title: Two modules never depend both ways
  @requirement LIB-00001-1
  for module m, module n
  where m != n
  never module-depends(m, n),
        module-depends(n, m)
}

definition depends(a, b) {
  if a uses b
  if m in a, m owns b
  if m in a, m refs b
  if m in a, s behaviour-of m, s uses b
}

rule store-calls-no-billing {
  title: No message of the store calls billing
  @requirement REQ-00002-1
  for message m in shop@store
  never m calls f, f in shop@billing
}

state Cart {
  Open       if status = Open, not cancelled
  Paid       if status = Paid, not cancelled
  Shipped    if status = Shipped
  Cancelled  if cancelled, status != Shipped
  Open -> Paid       on checkout
  Paid -> Shipped    on ship
  Open -> Cancelled  on cancel
}

rule store-requirements-satisfied {
  title: Every requirement of the store is satisfied
  @requirement REQ-00003-1
  for requirement r in shop@store
  holds x satisfies r
}
