structure: Interfaces

package shop {
  interface shell {
    kind: cli
    operation buy(item: string) -> int {
      command: shop buy <item>
      exit 0: default
      exit 2: declined
    }
  }

  sequence shell::buy {
    return 0
  }

  sequence shell::buy / declined {
    return 2
  }
}

diagram IF_Shell {
  kind: interface
  include: shop@shell
}

package shop {
  module store {
    class Line {
      + price: int
    }

    class Catalogue {
      + priceOf(item: string) -> int
    }

    class Coupon {
      + percent: int
    }

    class Cart {
      - owns lines: list<Line>
      - refs catalogue: Catalogue
      + apply(coupon: Coupon) -> none
      + total() -> int
    }
  }
}

  kind: class
  include: shop@store::*
