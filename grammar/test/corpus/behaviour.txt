package shop {

      + reserve(item: string) -> none

behaviour: One call inside another

package shop {
  module store {
    class Card {
      + pay(amount: int) -> bool
    }

    class Line {
      + subtotal() -> int
    }

    class Cart {
      - owns lines: list<Line>
      + total() -> int
      + checkout(card: Card) -> bool
    }

    sequence Cart::total {
      local line: Line
      loop each line {
        self -> line.subtotal() -> part
      }
      return sum
    }

    sequence Cart::checkout {
      -> total() -> amount
      self -> card.pay(amount) -> paid
      return paid
    }
  }
}

diagram SEQ_Checkout {
  kind: sequence
  include: shop@store::Cart::checkout
  depth: *
}

behaviour: Failures

package shop {
  module store {
    class Stock {
      + reserve(item: string) -> none fails SoldOut
    }

    class Cart {
      - stock: Stock
      + add(item: string) -> bool
    }

    sequence Stock::reserve {
      opt nothing is left / empty {
        fail SoldOut
      }
    }

    sequence Cart::add {
      attempt {
        self -> stock.reserve(item)
      } on SoldOut {
        return false
      }
      return true
    }
  }
}

  kind: sequence
  include: shop@store::Cart::add
  depth: *
}
